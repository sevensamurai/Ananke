name: etl-pipeline
models:
  planner:
    provider: openai
    model:    gpt-4.1-mini
  analyst:
    provider: openai
    model:    gpt-4.1-mini
# Jon declarations. "type: code" jobs must use Bind().
jobs:
  plan:
    type: agent
    model: planner
    system_prompt: |
      You are an ETL planning agent...
  fetch_a:
    type: code
  transform_a:
    type: agent
    model: analyst
    system_prompt: |
      You are a data transformation agent...
 
# Topology — DSL syntax.
connections:
  - plan -> fork(fetch_a, fetch_b)
  - fetch_a -> transform_a
  - fetch_b -> transform_b
  - join(transform_a, transform_b) -> combine
  - combine -> End

var scaffold = WorkflowScaffold.Parse<MyState>
    ("etl-pipeline", dsl);
var workflow  = scaffold
    .Bind("plan",    planJob)
    .Bind("fetch_a", fetchAJob)
    .Bind("fetch_b", fetchBJob)
    .Bind("combine", combineJob)
    .BindMerge("combine", branches => Merge(branches))
    .Build();
